function nextOccurrence(d,freq){

  const dt=new Date(d);

  if(freq==='weekly')dt.setDate(dt.getDate()+7);

  else if(freq==='biweekly')dt.setDate(dt.getDate()+14);

  else if(freq==='monthly')dt.setMonth(dt.getMonth()+1);

  else if(freq==='semimonthly'){const day=dt.getDate();if(day<15)dt.setDate(15);else{dt.setMonth(dt.getMonth()+1);dt.setDate(1);}}

  return dt;

}

  const end=new Date(start);end.setDate(end.getDate()+days);

  const out=[];

}

function buildForecast(days=35){

  const t=today(),startDate=new Date(t.y,t.m,t.day);

  const events=[];

  for(let mo=0;mo<2;mo++){

    const d=new Date(startDate);d.setMonth(d.getMonth()+mo);

    S.bills.forEach(b=>{

      const dt=new Date(d.getFullYear(),d.getMonth(),b.day);

      if(mo===0&&b.paid)return;

      const diff=Math.round((dt-startDate)/86400000);

      if(diff>=0&&diff<days)events.push({diff,label:b.name,amt:-b.amt,type:'bill'});

    });

  }

  expandIncomeDates(days).forEach(occ=>{

    const diff=Math.round((occ.date-startDate)/86400000);

    if(diff>=0&&diff<days)events.push({diff,label:occ.name,amt:occ.amt,type:'income'});

  });

  events.sort((a,b)=>a.diff-b.diff);

  const dayEvents={};

  events.forEach(e=>{if(!dayEvents[e.diff])dayEvents[e.diff]=[];dayEvents[e.diff].push(e);});

  let running=S.balance;

  const points=[];

  for(let i=0;i<days;i++){

    if(dayEvents[i])dayEvents[i].forEach(e=>running=Math.round((running+e.amt)*100)/100);

    const dt=new Date(startDate);dt.setDate(dt.getDate()+i);

    points.push({day:i,date:new Date(dt),balance:running,events:dayEvents[i]||[]});

  }

  return{points,events};

}
